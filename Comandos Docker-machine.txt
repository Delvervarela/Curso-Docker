Secuencia para conectar dos contenedores por medio de ambassadores

## Instalar docker machine
choco install docker-machine

## Cracion de la Vm de redis
docker-machine create -d virtualbox redis-host

## Creacion de la vm de identidock
docker-machine create -d virtualbox identidock-host

## Creacion del contenedor real-redis
eval $(docker-machine env redis-host)
docker run -d --name real-redis redis:3

## Creacion del contenedor ambassador para redis
docker run -d --name real-redis-ambassador -p 6379:6379 --link real-redis: real-redis amount/ambassador 

## ## Creacion del contenedor redis-ambassador
eval $(docker-machine env identidock-host)
docker run -d --name redos_ambassador --expose 6379 -e REDIS_PORT_6379_TCP=tcp://$(docker-machine ip redis-host):6379 amouat/ambassador

##
docker run -d --name dnmonster amouat:dnmonster:1.0

##
docker run -d --link dnmonster:dnmonster --link redis_ambassador:redis -p 80:9090 amouat/identidock:1.0

## Curl para hacer una peticion a identidock
curl $(docker-machine ip identidock-host)
